%%% IMAGES (ALTERNATIVE TEXT AND DECORATIVE IMAGES)

% Every image in an accessible PDF needs one of two things:
%
%   * alternative text that says what the image conveys, for someone who
%     cannot see it, or
%   * a mark saying that the image is purely decorative, so that assistive
%     technology skips it.
%
% \cvimage[options]{file}{alternative text}
%     An image that carries information (a photograph of you, a logo
%     that identifies an organization, a chart). The ``options'' are
%     anything \includegraphics accepts, like height=1.5in. Describe
%     what the image shows or means in one short phrase or sentence, as
%     in
%
%       \cvimage[height=1.5in]{my_picture}{Photograph of Your Name}
%
% \decorativegraphics[options]{file}
%     An image that adds nothing to the content (a flourish, a
%     background, a picture that repeats text next to it).
%
% Prefer these two commands to a bare \includegraphics, which would
% leave a figure without alternative text in a tagged PDF.
%
% Where either command is used in the template below, it is commented
% out, since the template ships with no image files.
\usepackage{graphicx}
% Very old graphicx releases have no ``alt'' or ``artifact'' keys; accept
% and ignore them there so that these commands never break a build. They
% also do nothing on the DVI/dvips route and in a PDF that is not tagged.
\makeatletter
\@ifundefined{KV@Gin@alt}{\define@key{Gin}{alt}{}}{}
\@ifundefined{KV@Gin@artifact}{\define@key{Gin}{artifact}[true]{}}{}
\makeatother
\newcommand{\cvimage}[3][]{\includegraphics[alt={#3},#1]{#2}}
\newcommand{\decorativegraphics}[2][]{\includegraphics[artifact,#1]{#2}}

%%% HEADING AT TOP OF CURRICULUM VITAE

% The title (name) with a horizontal rule under it
% (optional argument typesets an object right-justified across from name
%  as well)
%
% Usage: \makeheading{name}
%        OR
%        \makeheading[right_object]{name}
%
% Place at top of document. It should be the first thing.
% If ``right_object'' is provided in the square-braced optional
% argument, it will be right justified on the same line as ``name'' at
% the top of the CV. For example:
%
%       \makeheading[\emph{Curriculum vitae}]{Your Name}
%
% will put an emphasized ``Curriculum vitae'' at the top of the document
% as a title. Likewise, a picture could be included:
%
%   \makeheading[{\cvimage[height=1.5in]{my_picture}{Photograph of Your Name}}]{Your Name}
%
% the picture will be flush right across from the name. For this example
% to work, make sure the extra set of curly braces is included. Use
% \cvimage (with alternative text) for a picture that matters, and
% \decorativegraphics for one that does not (see ``IMAGES'' above).
\newcommand{\makeheading}[2][]%
        {\hspace*{-\marginparsep minus \marginparwidth}%
         \begin{minipage}[t]{\textwidth+\marginparwidth+\marginparsep}%
             {\large \bfseries #2 \hfill #1}\\[-0.15\baselineskip]%
                 \rule{\columnwidth}{1pt}%
         \end{minipage}}

%%% SECTION HEADINGS

% The section headings. Flush left in small caps down pseudo-margin.
%
% Usage: \section{section name}
%
% The title is typeset in a zero-height box in the margin, and the text
% that follows starts on the same line. In a tagged PDF, the same
% commands that the kernel's own heading code uses (the ``sec'' tagging
% sockets, which do nothing when tagging is off) mark the title as a
% real heading, so that it appears in the document's structure and its
% bookmarks rather than as loose text.
\setcounter{secnumdepth}{0}% section titles are not numbered
\makeatletter
\renewcommand{\section}[1]{\pagebreak[3]%
    \vspace{1.3\baselineskip}%
    \ifdefined\UseTaggingSocket
        \UseTaggingSocket{sec/end}{1}%
        \UseTaggingSocket{sec/begin}{{1}{tag=\UseStructureName{sec/1}}}%
    \fi
    \phantomsection\addcontentsline{toc}{section}{#1}%
    \begingroup
    \ifdefined\UseTaggingSocket
        \UseTaggingSocket{sec/title/init}{1}%
    \fi
    \noindent\llap{\scshape\smash{\parbox[t]{\marginparwidth}{\hyphenpenalty=10000\raggedright #1}}}%
    \vspace{-\baselineskip}\par
    \endgroup}
\makeatother

%%% LISTS

% THE CV LIST: cvlist
%
%   \begin{cvlist}              % tight: details under an entry
%       \item ...               % an item with a bullet
%       \entry ...              % an item with no bullet
%       \marker{alt text}{$\dagger$} ...
%                               % an item with your own marker symbol
%   \end{cvlist}
%
%   \begin{cvlist}[loose]       % space between entries, and a page break
%       \entry ...              % is preferred before the list
%   \end{cvlist}
%
% Lists nest (up to three deep), and ``tight'' and ``loose'' lists can be
% mixed. Use a ``loose'' list for the entries of a section (jobs, courses,
% awards), and a ``tight'' list for the details under each entry.
%
% \marker{alt text}{symbol} is for symbols that stand in for a bullet.
% The alternative text (``dash'', ``star'', and so on) is what a screen
% reader speaks in a tagged PDF.
\newcommand{\entry}{\item[{}]}
\newcommand{\marker}[2]{\item[\mathalt{#1}{#2}]}

% --- Skip over the rest of this section unless you want to change how
% --- the lists are built.
%
% NOTE: \entry is \item[{}] rather than \item[] because tagged lists
% treat an empty \item[] as a request for the default label.
%
% Lists are built with enumitem's \newlist and configured with keys and
% kernel hooks only (no patching of \item or \end...), which keeps them
% valid when the PDF is tagged. Tagged lists accept plain lengths in
% \setlist, but not enumitem's ``leftmargin=*'', calc expressions, or the
% before/after keys, so those are replaced by a measured length
% (\listindent) and by hooks that run before and after the list.
%   * loose lists prefer a page break before the list begins (the
%     \pagebreak[2] in a hook that runs before the list), so the first
%     item is not separated from the line that precedes it
%   * loose lists remove some of the space that follows them
% (The width is measured when the document starts, in the body font,
% because the preamble's font is not the one the lists are typeset in.)
% (\bibbulletindent is the left margin of a bibsection list that has a
% bullet; the bullet hangs in the space that bibsection's negative item
% indent opens up.)
\newlength{\listindent}
\newlength{\bibbulletindent}
\AtBeginDocument{{\normalfont\normalsize
    \settowidth{\listindent}{\enskip\textbullet}%
    \global\listindent=\dimexpr\listindent+\labelsep\relax
    \global\bibbulletindent=\dimexpr\listindent+\bibhang\relax}}
\newcommand*{\listendspace}[1]{\AddToHook{env/#1/after}{\vspace{-0.6\baselineskip}}}
\newcommand*{\listbreakbefore}[1]{\AddToHook{env/#1/before}{\pagebreak[2]}}

% The two kinds of list that cvlist chooses between.
\newlist{cvlisttight}{itemize}{3}
    \setlist[cvlisttight]{label=\enskip\textbullet,leftmargin=\listindent,nosep}
\newlist{cvlistloose}{itemize}{3}
    \setlist[cvlistloose]{label=\enskip\textbullet,leftmargin=\listindent}
    \listendspace{cvlistloose}
    \listbreakbefore{cvlistloose}
\newcommand*{\cvlistloosename}{loose}
\NewDocumentEnvironment{cvlist}{O{}}
    {\def\cvlistdensity{#1}%
     \ifx\cvlistdensity\cvlistloosename
         \begin{cvlistloose}%
     \else
         \begin{cvlisttight}%
     \fi}
    {\ifx\cvlistdensity\cvlistloosename
         \end{cvlistloose}%
     \else
         \end{cvlisttight}%
     \fi}

%%% ALTERNATIVE TEXT FOR FORMULAS

% \mathalt{alt text}{math} typesets ``math'' as usual and, in a tagged
% PDF, gives that formula the alternative text ``alt text'' (what a screen
% reader speaks) in place of the default description built from its TeX
% source. Use it for symbols used as markers or bullets, as in
%
%   \marker{dagger}{$\dagger$} ...
%
% Without PDF tagging, it just typesets the math. (The tagged version
% uses a tagging socket of LaTeX's math code, so it is skipped where the
% socket does not exist.)
\newcommand{\mathalt}[2]{#2}
\ifdefined\NewTaggingSocketPlug
\ExplSyntaxOn
\cs_if_exist:NT \l__math_content_alt_tl
  {
    \tl_new:N \l_cv_mathalt_tl
    \NewTaggingSocketPlug { math/content } { cv-alt }
      {
        \tl_set:Ne \l__math_content_alt_tl
          {
            \tl_if_empty:NTF \l_cv_mathalt_tl
              { \l__math_content_template_tl }
              { \l_cv_mathalt_tl }
          }
        \tl_set:Ne \l__math_content_AF_source_tl { \l__math_texsource_template_tl }
        \tl_set:Nn \l__math_content_AF_mathml_tl { }
        \tl_set:Nn \l__math_content_actual_tl { }
      }
    \AssignTaggingSocketPlug { math/content } { cv-alt }
    \RenewDocumentCommand \mathalt { m m }
      { \group_begin: \tl_set:Nn \l_cv_mathalt_tl {#1} #2 \group_end: }
  }
\ExplSyntaxOff
\fi
